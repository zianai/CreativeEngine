/** Focused local collection of this new native image only; no replay/recovery/export regression. */
import assert from 'node:assert/strict';
import {readFileSync,writeFileSync,mkdirSync,copyFileSync} from 'node:fs';
import {join,resolve} from 'node:path';
import {Core} from '/Users/a/workspaces/AForeign Trade/validation-coordinator/worktrees/creative-engine/packages/core/service.ts';
import {OracleProvider} from '/Users/a/workspaces/AForeign Trade/validation-coordinator/worktrees/creative-engine/packages/providers/oracle/index.ts';
import {sha} from '/Users/a/workspaces/AForeign Trade/validation-coordinator/worktrees/creative-engine/packages/core/validation.ts';
import {validatedRaster} from '/Users/a/workspaces/AForeign Trade/validation-coordinator/worktrees/creative-engine/packages/providers/files.ts';
const folder=resolve(process.argv[2]),target=JSON.parse(readFileSync(join(folder,'target.json'),'utf8'));
const provider=new OracleProvider({stateDir:join(target.data_root,'provider-receipts/oracle')});
const core=new Core(target.data_root,new Map([['oracle',provider]]));
const out=join(folder,'new-image');mkdirSync(out,{recursive:true});
try{
 const task=core.require('task',target.task_id);assert.equal(task.status,'review_pending');assert.equal(task.attempt_count,1);
 const asset=core.require('asset',task.asset_id),receipt=await provider.getResult(task.attempt_id);assert(receipt&&!receipt.mock);assert.equal(asset.review_status,'pending');
 const images=[];
 for(const [name,path,meaning] of [
  ['oracle-downloaded.png',join(folder,'oracle-home/generated',task.attempt_id+'.png'),'Oracle actual downloaded bytes'],
  ['provider-normalized.png',receipt.path,'Provider-normalized bytes'],
  ['native-review.png',asset.path,'Native asset after one application concept label']
 ]){const raster=await validatedRaster(path);copyFileSync(path,join(out,name));images.push({file:name,sha256:sha(readFileSync(path)),...raster,meaning});}
 const result={status:'new_image_collected',source_sha:target.source_sha,task_id:task.id,attempt_id:task.attempt_id,asset_id:asset.id,session_id:receipt.sessionId,review_status:'pending',images,visual_checks:{repeated_flexible_pleats:'pending_image_inspection',no_barrel_piano_or_pin_hinges:'pending_image_inspection',one_compartment_partly_folded:'pending_image_inspection',downloaded_image_has_no_generated_text:'pending_image_inspection',review_image_has_one_concept_label:'pending_image_inspection'},model_verified:false,human_approval:false,api_requests_added:0,scope:'Only this new output; no previous passing regression repeated'};
 writeFileSync(join(out,'result.json'),JSON.stringify(result,null,2)+'\n');
 writeFileSync(join(out,'receipt.json'),JSON.stringify(receipt,null,2)+'\n');
 console.log(JSON.stringify(result));
}finally{core.close();await provider.close();}
